\begin{algorithm}[H]
    \SetKwFunction{FInsertionSort}{InsertionSort}

    \Fn{\FInsertionSort{$A$}}{
        $n \gets \FLength{A}$\;
        \bl
        \For{$i \gets 1$ \KwTo $n - 1$}{
            $j \gets i$\;
            \bl
            \While{$j > 0$ \textbf{and} $A[j-1] > A[j]$}{
                \FSwap{$A[j-1], A[j]$}\;
                $j \gets j - 1$\;
            }
        }
    }
\end{algorithm}
