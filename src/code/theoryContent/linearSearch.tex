\begin{algorithm}[H]
    \SetKwFunction{FLinearSearch}{LinearSearch}

    \Fn{\FLinearSearch{$A, target$}}{
        $n \gets \FLength{A}$\;
        \bl
        \For{$i \gets 0$ \KwTo $n - 1$}{
            \If{$A[i] == target$}{
                \Return{i}\;
            }
        }
        \Return{-1}\;
    }
\end{algorithm}
