\begin{algorithm}[H]
    \SetKwFunction{FBubbleSort}{BubbleSort}

    \Fn{\FBubbleSort{$A$}}{
        $n \gets \FLength{A}$\;
        \bl
        \For{$i \gets 0$ \KwTo $n - 1$}{
            \For{$j \gets 0$ \KwTo $n - i - 1$}{
                \If{$A[j] > A[j + 1]$}{
                    \FSwap{$A[j], A[j + 1]$}\;
                }
            }
        }
    }
\end{algorithm}
